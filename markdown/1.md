这道题目由于操作次数和网格规模较大，直接对每个骰子进行暴力搜索显然会超时。因此我们需要充分挖掘骰子排列独立性的特点，利用“高度事件点分解”的方法将同一种暴露面状态的骰子合并计算。

### 题目剖析
1. **独立旋转：** 每颗骰子的旋转互不影响。对于任意坐标的某个骰子，我们只需知道它的哪些面被暴露在外部，然后将它单独旋转到**能使这些暴露面点数之和最大**的合法状态即可。
2. **暴露条件的规律性：** 考虑坐标 $(i, j)$ 处一座高度为 $h$ 的骰子塔，对其中的任意第 $z$ 层（$1 \le z \le h$）的骰子：
   - **顶面 (Top)** 暴露，当且仅当 $z = h$。
   - **底面 (Bottom)** 暴露，当且仅当 $z = 1$。
   - **东南西北四个侧面 (N, S, W, E)** 暴露，当且仅当它的对应侧面相邻柱子的高度 $< z$。
3. **状态压缩与预处理：**
   骰子的一共有 6 个面，一共有 $2^6 = 64$ 种暴露情况（可以使用一个 0 ~ 63 的掩码位 `mask` 表示）。
   由于骰子的基本镜像手性不可变，符合标准规则的骰子在 3D 空间中一共有 $6 \times 4 = 24$ 种合法的旋转状态。我们完全可以在程序运行的最开始，花 $O(1)$ 的极小时间将所有掩码对应的**最大得分情况提前打表预处理**好。
4. **离散化区间分段（极速查询）：**
   对单根柱子而言，由于最大高度 $h$ 可能高达 $10^9$，不能逐层遍历。我们发现随着 $z$ 从 $1$ 递增至 $h$，上面所述的面暴露条件只有在特定的一些“临界点”才会发生变化。
   这些临界点包含：$1, 2, h, h+1$ 以及**相邻四周四个柱子的高度 $H_k + 1$**。
   将这至多 8 个临界点进行排序、去重，它们会把所有的 $z$ 自动切分成至多 7 个连续的高速跳跃区间。在每一个区间内所有骰子的 `mask` 必定是完全一致的！直接乘上该区间的长度即可。

### 极致精简 C++ 代码

```cpp
#pragma GCC optimize("O3,unroll-loops")
#include <bits/stdc++.h>
using namespace std;

int max_score[64];

// O(1) 预计算：24种骰子合法旋转姿态下的各掩码得分最优解
void precompute() {
    memset(max_score, 0, sizeof(max_score));
  
    // 逆时针环绕邻居表：给定顶面 T，返回周围 4 个侧面的逆时针顺序
    // 下标为 T, 对应序列表示 {North, West, South, East} 依次能担任的角色
    int CCW[7][4] = {
        {0, 0, 0, 0},
        {2, 3, 5, 4}, // 顶为 1
        {1, 3, 6, 4}, // 顶为 2
        {1, 5, 6, 2}, // 顶为 3
        {1, 2, 6, 5}, // 顶为 4
        {1, 4, 6, 3}, // 顶为 5
        {2, 4, 5, 3}  // 顶为 6
    };

    for (int T = 1; T <= 6; ++T) {
        for (int k = 0; k < 4; ++k) {
            int N = CCW[T][k];
            int W = CCW[T][(k + 1) % 4];
            int S = 7 - N;
            int E = 7 - W;
            int B = 7 - T;
            int V[6] = {T, B, N, S, W, E}; // 映射下标：0:Top, 1:Bottom, 2:North, 3:South, 4:West, 5:East
          
            for (int mask = 0; mask < 64; ++mask) {
                int score = 0;
                for (int i = 0; i < 6; ++i) {
                    if ((mask >> i) & 1) score += V[i];
                }
                max_score[mask] = max(max_score[mask], score);
            }
        }
    }
}

void solve() {
    int n, m;
    cin >> n >> m;
  
    // 采用 1D 展平存储二维网格，提供最高级别的 Cache 加速命中率
    vector<long long> grid(n * m);
    for (int i = 0; i < n * m; ++i) {
        cin >> grid[i];
    }
  
    long long total_score = 0;
  
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < m; ++j) {
            long long h = grid[i * m + j];
            if (h == 0) continue;
          
            // 获取并截断四周邻柱高度（超过 h 的遮蔽效果和等于 h 是一样的）
            long long H_neighbors[4] = {
                i > 0 ? grid[(i - 1) * m + j] : 0,
                i < n - 1 ? grid[(i + 1) * m + j] : 0,
                j > 0 ? grid[i * m + (j - 1)] : 0,
                j < m - 1 ? grid[i * m + (j + 1)] : 0
            };
          
            for (int k = 0; k < 4; ++k) {
                if (H_neighbors[k] > h) H_neighbors[k] = h;
            }
          
            // 组装关键切割点数组 Z（至多 8 个边界点）
            long long Z[8];
            Z[0] = 1;
            Z[1] = 2;
            Z[2] = h;
            Z[3] = h + 1;
            Z[4] = H_neighbors[0] + 1;
            Z[5] = H_neighbors[1] + 1;
            Z[6] = H_neighbors[2] + 1;
            Z[7] = H_neighbors[3] + 1;
          
            sort(Z, Z + 8);
            int unique_size = unique(Z, Z + 8) - Z;
          
            // O(1) 离散化区间打分法
            for (int k = 0; k + 1 < unique_size; ++k) {
                long long L = Z[k];
                long long R = Z[k + 1];
              
                if (L > h) break; // 区间完全超出柱子实体高度，结束计算
              
                long long z = L; // 选取区间内任意一点推导遮蔽 mask
                int mask = 0;
              
                if (z == h) mask |= (1 << 0);
                if (z == 1) mask |= (1 << 1);
                if (z > H_neighbors[0]) mask |= (1 << 2);
                if (z > H_neighbors[1]) mask |= (1 << 3);
                if (z > H_neighbors[2]) mask |= (1 << 4);
                if (z > H_neighbors[3]) mask |= (1 << 5);
              
                // 将 “本连续同态段内拥有的骰子数量” 乘以 “当前姿态预计算极限得分” 累加
                total_score += (R - L) * max_score[mask];
            }
        }
    }
  
    cout << total_score << "\n";
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    precompute();
    int t;
    if (cin >> t) {
        while (t--) solve();
    }
    return 0;
}
```
