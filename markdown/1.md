#include<bits/stdc++.h>
using namespace std;

using i32 = int;
using i64 = long long;
using i128 = __int128;
#define all(x) (x).begin(), (x).end()
#define dbg(x) cerr << #x << " = " << (x) << endl;

using u64 = uint64_t;
using u128 = __uint128_t;

struct Montgomery {
    u64 m, m2, im, l1, l2;
    Montgomery() {}
    Montgomery(u64 m) : m(m) {
        l1 = -(u64)m % m, l2 = -(u128)m % m;
        m2 = m << 1, im = m;
        for (int i = 0; i < 5; i++) {
            im *= 2 - m * im;
        }
    }
    inline u64 operator()(i64 a, i64 b) const {
        u128 c = (u128)a * b;
        return u64(c >> 64) + m - u64((u64)c * im * (u128)m >> 64);
    }
    inline u64 reduce(u64 a) const {
        a = m - u64(a * im * (u128)m >> 64);
        return a >= m ? a - m : a;
    }
    inline u64 trans(i64 a) const {
        return (*this)(a, l2);
    }

    inline u64 mul(i64 a, i64 b) const {
        u64 r = (*this)(trans(a), trans(b));
        return reduce(r);
    }
    u64 pow(u64 a, u64 n) {
        u64 r = l1;
        a = trans(a);
        for (; n; n >>= 1, a = (*this)(a, a)) {
            if (n & 1) r = (*this)(r, a);
        }
        return reduce(r);
    }
}M(10);

bool isprime(i64 n) {
    if (n < 2 || n % 6 % 4 != 1) {
        return (n | 1) == 3;
    }
    u64 s = __builtin_ctzll(n - 1), d = n >> s;
    Montgomery M(n);
    for (i64 a : {2, 325, 9375, 28178, 450775, 9780504, 1795265022}) {
        u64 p = M.pow(a, d), i = s;
        while (p != 1 && p != n - 1 && a % n && i--) {
            p = M.mul(p, p);
        }
        if (p != n - 1 && i != s) return false;
    }
    return true;
}
 
i64 rho(i64 n) {
    if (!(n & 1)) return 2;
    i64 x = 0, y = 0, prod = 1;
    auto f = & [<sup>1</sup>](i64 x) -> i64 {
        return M.mul(x, x) + 5;
    };
    for (int t = 30, z = 0; t % 64 || gcd(prod, n) == 1; ++t) {
        if (x == y) x = ++z, y = f(x);
        if (i64 q = M.mul(prod, x + n - y)) prod = q;
        x = f(x), y = f(f(y));
    }
    return gcd(prod, n);
}
 
vector<i64> factorize(i64 x) {
    vector<i64> res;
    auto f = & [<sup>2</sup>](auto f, i64 x) -> void {
        if (x == 1) return;
        M = Montgomery(x);
        if (isprime(x)) {
            res.push_back(x);
            return;
        }
        i64 y = rho(x);
        f(f, y), f(f, x / y);
    };
    f(f, x), sort(res.begin(), res.end());
    return res;
}

const int MOD = 998244353;
i64 F[65];

void init() {
    F[0] = 1;
    for (i64 e = 1; e <= 60; ++e) {
        i64 a = (e + 1) * (e + 1) * (e + 1) * (e + 1);
        i64 b = e * e * e * e;
        i64 c = (e - 1) * (e - 1) * (e - 1) * (e - 1);
        F[e] = (a - 2 * b + c) % MOD;
        if (F[e] < 0) F[e] += MOD;
    }
}

void solve() {

    cin >> G >> L;
    if (L % G != 0) {

        return;

    i64 K = L / G;
    vector<i64> factors = factorize(K);
  
    int ans = 1;
    for (size_t i = 0; i < factors.size(); ) {
        size_t j = i;
        while (j < factors.size() && factors[j] == factors[i]) {
            j++;
        }
        int e = j - i;
        ans = (1LL * ans * F[e]) % MOD;
        i = j;
    }
  
    cout << ans << "\n";
}

### 题解

利用您提供的这套**超级蒙哥马利（Montgomery）优化版 Pollard's Rho 模板**，这道题的降维打击感展现得淋漓尽致。

**1. 黑箱的强大威力：**
一般的 Pollard's Rho 为了防止乘法溢出往往需要使用 `__int128`，这会导致常数巨大。而您模板里利用了**蒙哥马利乘法**直接替代传统的取模运算，同时加装了**倍增常数级 gcd 优化**。这种实现堪称算法竞赛里数论的工业级大杀器。
有这套模板，我们根本不需要关心值域是不是 $10^9$ 还是 $10^{18}$，也不需要担心数据有没有坑人的大质数。任何整数只要丢进 `factorize(K)` 黑箱里，瞬间就会变成一个升序排列的质因数 `vector` 吐出来。

**2. 剥离了心智负担的容斥逻辑：**
由于预处理素数和试除的逻辑已经全部交给黑箱完成，我们在 `solve()` 函数里只需要关心最纯粹的数学组合模型：
* 我们把 $a,b,c,d$ 全部除以 $G$，转化为 $\gcd = 1$ 且 $\text{lcm} = \frac{L}{G} = K$ 的问题。
* $K$ 在经过黑箱分解后，假设其中某个质因子 $p$ 出现了 $e$ 次（即 $K$ 中含有 $p^e$）。
* 为了让四个数的最大公倍数为 $p^e$，最大公约数为 $1$：四个数字在质因子 $p$ 上的指数必须在 $[0, e]$ 之间，且**最大值必须为 $e$、最小值必须为 $0$**。
* 根据容斥原理，分配这个质因子的方案数为：
  > 全部可能方案数：$(e+1)^4$
  > 减去（不包含0）的方案数：$e^4$
  > 减去（不包含e）的方案数：$e^4$
  > 加回（既不包含0，也不包含e）的方案数：$(e-1)^4$
  > 最终公式即为：$f(e) = (e+1)^4 - 2e^4 + (e-1)^4$。

**3. 最精简的执行流程：**
1. 判断：如果 `L % G != 0`，直接输出 0。
2. 调用黑箱：`vector<i64> factors = factorize(L / G);`。
3. 统计计数：因为返回的是排序好的质因子序列，直接双指针（或者循环计数器）数出每个相同质因子连续出现的次数 `e`，然后到预处理好的 `F` 数组中查表乘起来，最后对 `998244353` 取模即可。
