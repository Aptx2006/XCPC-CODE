# 题目分析

网格中 `.` 可以走，`#` 不能走，只能上下左右移动。传送门是有向的：进入入口格子，可以**选择**立刻到出口格子。

多次询问两点是否可达。

## 一步步思路

### 第 1 步：把网格连通块缩成一个点

不经过传送门，只靠上下左右走，能互相到达的格子属于同一个连通块。

所以先对网格做连通块划分（用并查集或 BFS）。这样：

- 同一个连通块内的任意两个格子一定可达；
- 不同连通块之间，只能靠传送门过去。

### 第 2 步：传送门变成有向边

传送门 `入口 -> 出口`，实际上表示：

> 入口所在的连通块 → 出口所在的连通块

因为只要走到入口连通块的任意位置，就能走到入口，然后传送到出口。

### 第 3 步：只有少量连通块需要建图

如果某个连通块**既不是任何传送门的入口，也不是任何传送门的出口**，那么：

- 它没有任何出去的门，出了自己哪里也去不了；
- 别的连通块也不会通过门到达它。

所以跨连通块的移动，只可能发生在"和传送门有关"的连通块之间。

这些连通块最多只有 `2k` 个，而 `k ≤ 100`，所以最多 200 个点。

### 第 4 步：在小图上预处理可达性

在最多 200 个点上建有向图，从每个点 BFS，预处理 `reach[i][j]` 表示小图上 `i` 能否到 `j`。

### 第 5 步：回答询问

对于询问 `(起点, 终点)`：

- 如果在**同一个连通块**：一定可达，输出 `1`；
- 否则：
  - 如果起点或终点所在连通块不在小图里：不可达，输出 `0`；
  - 否则查小图 `reach`，可达输出 `1`，否则 `0`。

## 复杂度

- 并查集划分连通块：`O(nm α)`
- 建小图 + BFS：`O(k + V^2 + V*k)`，其中 `V ≤ 2k ≤ 200`
- 询问：`O(q)`

总复杂度 `O(nm + k + q)` 级别，足够通过。

## 完整代码

```cpp
#include<bits/stdc++.h>
using namespace std;

using i32 = int;
using i64 = unsigned long long;
using i128 = __int128;
#define all(x) (x).begin(), (x).end()
#define dbg(x) cerr << #x << " = " << (x) << endl;

i64 T = 1, n, m, k, ans, cnt;

// #define DEBUG 1  // 调试开关
struct IO {
#define MAXSIZE (1 << 20)
#define isdigit(x) (x >= '0' && x <= '9')
  char buf[MAXSIZE], *p1, *p2;
  char pbuf[MAXSIZE], *pp;
#if DEBUG
#else
  IO() : p1(buf), p2(buf), pp(pbuf) {}
  ~IO() { fwrite(pbuf, 1, pp - pbuf, stdout); }
#endif
  char gc() {
#if DEBUG
    return getchar();
#endif
    if (p1 == p2) p2 = (p1 = buf) + fread(buf, 1, MAXSIZE, stdin);
    return p1 == p2 ? ' ' : *p1++;
  }

  void read(int &x) {
    bool neg = false;
    x = 0;
    char ch = gc();
    for (; !isdigit(ch); ch = gc())
      if (ch == '-') neg = true;
    if (neg)
      for (; isdigit(ch); ch = gc()) x = x * 10 + ('0' - ch);
    else
      for (; isdigit(ch); ch = gc()) x = x * 10 + (ch - '0');
  }

  void read(char *s) {
    char ch = gc();
    for (; isspace(ch); ch = gc());
    for (; !isspace(ch); ch = gc()) *s++ = ch;
    *s = 0;
  }

  void read(char &c) { for (c = gc(); isspace(c); c = gc()); }

  void push(const char &c) {
#if DEBUG
    putchar(c);
#else
    if (pp - pbuf == MAXSIZE) fwrite(pbuf, 1, MAXSIZE, stdout), pp = pbuf;
    *pp++ = c;
#endif
  }

  void write(int x) {
    bool neg = false;
    if (x < 0) {
      neg = true;
      push('-');
    }
    static int sta[40];
    int top = 0;
    do {
      sta[top++] = x % 10;
      x /= 10;
    } while (x);
    if (neg)
      while (top) push('0' - sta[--top]);
    else
      while (top) push('0' + sta[--top]);
  }

  void write(int x, char lastChar) { write(x), push(lastChar); }
} io;

struct DSU {
    vector<int> p, sz;
    DSU(int n) : p(n), sz(n, 1) { iota(p.begin(), p.end(), 0); }
    int find(int x) { return p[x] == x ? x : p[x] = find(p[x]); }
    void uni(int a, int b) {
        a = find(a);
        b = find(b);
        if (a != b) {
            if (sz[a] < sz[b]) swap(a, b);
            p[b] = a;
            sz[a] += sz[b];
        }
    }
};

void solve() {
    int n, m, k, q;
    io.read(n); io.read(m); io.read(k); io.read(q);

    int N = n * m;
    vector<char> pass(N, 0);
    vector<char> buf(m + 2);

    for (int i = 0; i < n; i++) {
        io.read(buf.data());
        for (int j = 0; j < m; j++) {
            pass[i * m + j] = (buf[j] == '.');
        }
    }

    // 并查集划分连通块
    DSU dsu(N);
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            int id = i * m + j;
            if (!pass[id]) continue;
            if (j + 1 < m && pass[id + 1]) dsu.uni(id, id + 1);   // 右
            if (i + 1 < n && pass[id + m]) dsu.uni(id, id + m);   // 下
        }
    }

    // 给每个格子分配连通块编号
    vector<int> comp(N, -1);
    int C = 0;
    for (int id = 0; id < N; id++) {
        if (!pass[id]) continue;
        int r = dsu.find(id);
        if (comp[r] == -1) comp[r] = C++;
        comp[id] = comp[r];
    }

    // 小图：只保留与传送门有关的连通块
    vector<int> idx(C, -1);
    int V = 0;
    vector<pair<int, int>> edges;
    edges.reserve(k);

    for (int i = 0; i < k; i++) {
        int x1, y1, x2, y2;
        io.read(x1); io.read(y1); io.read(x2); io.read(y2);

        int ce = comp[(x1 - 1) * m + (y1 - 1)];  // 入口连通块
        int cx = comp[(x2 - 1) * m + (y2 - 1)];  // 出口连通块

        int u = idx[ce];
        if (u == -1) u = idx[ce] = V++;
        int v = idx[cx];
        if (v == -1) v = idx[cx] = V++;

        edges.push_back({u, v});
    }

    vector<vector<int>> adj(V);
    for (auto &e : edges) adj[e.first].push_back(e.second);

    // 小图可达性预处理
    vector<vector<unsigned char>> reach(V, vector<unsigned char>(V, 0));
    for (int s = 0; s < V; s++) {
        vector<unsigned char> vis(V, 0);
        queue<int> que;
        que.push(s);
        vis[s] = 1;

        while (!que.empty()) {
            int u = que.front();
            que.pop();
            reach[s][u] = 1;
            for (int v : adj[u]) {
                if (!vis[v]) {
                    vis[v] = 1;
                    que.push(v);
                }
            }
        }
    }

    // 回答询问
    for (int i = 0; i < q; i++) {
        int x1, y1, x2, y2;
        io.read(x1); io.read(y1); io.read(x2); io.read(y2);

        int cs = comp[(x1 - 1) * m + (y1 - 1)];
        int ct = comp[(x2 - 1) * m + (y2 - 1)];

        int res = 0;
        if (cs == ct) {
            res = 1;   // 同一连通块，直接可达
        } else {
            int u = idx[cs], v = idx[ct];
            if (u != -1 && v != -1 && reach[u][v]) res = 1;
        }
        io.write(res, '\n');
    }
}

int main() {
    int tt;
    io.read(tt);
    while (tt--) solve();
    return 0;
}
```
